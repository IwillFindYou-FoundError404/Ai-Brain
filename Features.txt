const AstraBrainManager = {
    activeDBIndex: 0,
    currentMongoDBUri: MONGO_CLUSTERS[0],
    mongoClient: null,

    // QUẢN LÝ KHÔNG GIAN BỘ NHỚ TRÊN RAM TẠM
    ramStorage: {
        primaryBrain: { version: "v1.0.0-stable", weights: {}, rules: {} }, 
        sandboxBrain: { version: "v1.0.1-sandbox", adjustedRules: {} }, 
        mistakeHistory: [],       // Mảng chứa các log báo lỗi văn bản siêu nhẹ
        maxMistakeThreshold: 50,  // Giới hạn khống chế 50 lỗi trên RAM chống tràn bộ nhớ
        maxBrainLimit: 307        
    },

    // BỘ ĐỊNH TUYẾN CHUYỂN VÙNG DATABASE KHI ĐẦY DUNG LƯỢNG
    async checkAndRouterDatabase() {
        try {
            let isCurrentClusterFull = false; // Thiết lập logic check thực tế từ DB stats
            
            if (isCurrentClusterFull) {
                if (this.activeDBIndex < MONGO_CLUSTERS.length - 1) {
                    this.activeDBIndex++;
                    this.currentMongoDBUri = MONGO_CLUSTERS[this.activeDBIndex];
                    
                    console.warn(`[AIH Guard] PHÁT HIỆN CLUSTER SẮP ĐẦY!`);
                    console.warn(`[System Router] Tự động chuyển cổng dữ liệu sang Tài khoản Google số ${this.activeDBIndex + 1}`);
                    
                    // Thực hiện logic tái kết nối kết nối driver:
                    // if (this.mongoClient) { await this.mongoClient.close(); }
                    // this.mongoClient = await MongoClient.connect(this.currentMongoDBUri);
                } else {
                    console.error("[AIH Danger] CẢNH BÁO: Hết sạch 3.5GB lưu trữ trên toàn bộ 7 tài khoản Google!");
                }
            }
        } catch (err) {
            console.error("[Database Router Error]", err);
        }
    },

    // THUẬT TOÁN GOM LỖI NUÔI NÃO SANDBOX (LEARNING ENGINE)
    learnFromMistake(agentName, issueType, rawInput, failedOutput, fixSuggestion) {
        const mistakeNode = {
            timestamp: new Date().toISOString(),
            agent: agentName,
            issue: issueType,
            inputContext: rawInput.substring(0, 100), // Khống chế độ dài chuỗi ký tự tránh leak RAM
            badResponse: failedOutput.substring(0, 100),
            evolutionFix: fixSuggestion
        };

        this.ramStorage.mistakeHistory.push(mistakeNode);

        // Thuật toán dọn rác mảng (LRU Cache) giữ dung lượng RAM cố định
        if (this.ramStorage.mistakeHistory.length > this.ramStorage.maxMistakeThreshold) {
            this.ramStorage.mistakeHistory.shift(); 
        }

        console.log(`[Astra Brain Learning] Ghi nhận Mini Agent [${agentName}] vấp ngã. Đã nạp bài học vào RAM Sandbox.`);
        this.refineSandboxBrain();
    },

    // CƠ CHẾ TỰ ĐỘNG HIỆU CHỈNH LOGIC NÃO ẢO SANDBOX
    refineSandboxBrain() {
        this.ramStorage.mistakeHistory.forEach(log => {
            // AI tự động cập nhật hướng sửa lỗi vào cấu trúc luật của não Sandbox
            this.ramStorage.sandboxBrain.adjustedRules[log.issue] = log.evolutionFix;
        });
    },

    // CHU TRÌNH THĂNG CẤP THÀNH BRAIN 2 HOÀN CHỈNH & ĐỒNG BỘ MÂY
    async promoteSandboxToBrain2() {
        try {
            console.log("[Brain Lifecycle] Đang chấm điểm độ ổn định của bộ não Sandbox...");
            let isSandboxStable = true; // Thuật toán tự đánh giá độ thông minh của Sandbox

            if (isSandboxStable) {
                // Thăng cấp đổi tên thành phiên bản tiến hóa mới, ghi đè thay thế bộ não cũ
                this.ramStorage.primaryBrain = { 
                    ...this.ramStorage.sandboxBrain, 
                    version: `v2.0.0-Evolved-${Date.now()}` 
                };
                
                // Sao lưu dữ liệu bộ não hoàn chỉnh này lên cụm MongoDB đang kết nối
                await this.syncBrainToCloud(this.ramStorage.primaryBrain);
                
                // Giải phóng RAM sạch sẽ sau khi thăng cấp thành công
                this.ramStorage.mistakeHistory = [];
                if (global.gc) global.gc(); 
                
                console.log("[Brain Lifecycle] Tiến hóa thành công! Đã giải phóng RAM, đưa Brain 2 lên Cloud.");
            } else {
                // HÀM ROLLBACK: Nếu gặp sự cố, xóa trực tiếp bộ não Sandbox lỗi, khôi phục cài đặt gốc
                this.ramStorage.sandboxBrain = { version: "v1.0.1-sandbox", adjustedRules: {} };
                console.warn("[Brain Lifecycle] Não Sandbox lỗi. Kích hoạt lệnh tự hủy để bảo vệ hệ thống.");
            }
        } catch (error) {
            console.error("[Brain Lifecycle Error]", error);
        }
    },

    async syncBrainToCloud(stableBrainData) {
        await this.checkAndRouterDatabase();
        // Lệnh đẩy dữ liệu JSON thực tế lên MongoDB Atlas:
        // await this.mongoClient.db("Astra").collection("active_brains").insertOne(stableBrainData);
        console.log(`[Cloud Backup] Đã đồng bộ cấu trúc bộ não lên MongoDB URI: ${this.currentMongoDBUri.substring(0, 45)}...`);
    }
};

## 1. Tối ưu hóa bộ nhớ RAM tạm thời (Runtime Memory)

* Sử dụng Stream thay vì Buffer để đọc/ghi tệp: Khi cần xử lý các file dữ liệu, nhật ký (Logs), hoặc file cấu hình lớn, tuyệt đối không dùng hàm đọc trọn gói (như fs.readFile). Việc này sẽ bắt RAM phải gánh nguyên dung lượng file đó. Thay vào đó, hãy dùng fs.createReadStream để chia nhỏ file thành từng khúc nhỏ (chunks) vài chục KB để xử lý cuốn chiếu, RAM sẽ luôn trống trải.
* Tránh bẫy biến toàn cục (Global Variables / Closures): Biến toàn cục hoặc các hàm lồng nhau giữ lại tham chiếu dữ liệu sẽ khiến bộ dọn rác (Garbage Collector) của Node.js không thể giải phóng vùng nhớ. Hãy cố gắng giới hạn biến bên trong các phạm vi hàm cục bộ (Local scope) để khi hàm chạy xong, RAM tự động được thu hồi.
* Thay thế mảng Object bằng cấu trúc dữ liệu nguyên bản (Buffer/TypedArrays): Nếu bạn cần lưu trữ hàng triệu ma trận số, tọa độ 3D hoặc id hệ thống, việc dùng mảng JSON thông thường ([{x:1, y:2}]) ngốn RAM gấp 5–10 lần so với việc dùng mảng kiểu dữ liệu thuần của JavaScript như Float32Array hay Int32Array.

## 2. Tối ưu hóa cấu trúc và dung lượng mã nguồn (Code Size)

* Sử dụng kỹ thuật nén và rút gọn mã (Minification & Bundling): Trước khi deploy code lên môi trường production (như Render), hãy chạy code qua các công cụ như Esbuild, Terser hoặc Webpack. Các công cụ này sẽ tự động xóa toàn bộ khoảng trắng, chú thích (comments), gộp nhiều file module và đổi tên các biến dài thành các ký tự siêu ngắn (ví dụ: activeDatabaseIndex thành a). Việc này có thể ép một thư mục code vài trăm KB xuống chỉ còn vài chục KB.
* Quản lý dependencies (Thư viện ngoài) chặt chẽ: Chỉ cài đặt các thư viện thực sự cần thiết. Hãy kiểm tra file package.json và loại bỏ các gói không dùng đến. Khi cài thư viện lớn, hãy tận dụng kỹ thuật Tree-shaking để chỉ import đúng hàm mình cần dùng (ví dụ: import { chunk } from 'lodash' thay vì bê nguyên cả thư viện lodash vào code).

## 3. Tối ưu hóa tệp dữ liệu lưu trữ (Storage Optimization)

* Nén dữ liệu trước khi lưu đĩa hoặc đẩy lên Cloud: Nếu bắt buộc phải lưu file hoặc gửi chuỗi dữ liệu JSON lớn lên Database, hãy dùng các thuật toán nén chuỗi có sẵn như Gzip hoặc Brotli (trong Node.js là module zlib). Một chuỗi ký tự kịch bản dài sau khi nãn có thể giảm tới 70% dung lượng, giúp tiết kiệm đáng kể băng thông mạng và không gian đĩa.
* Thiết lập vòng đời cho dữ liệu (Data TTL - Time To Live): Đối với các dữ liệu tạm thời như bộ nhớ đệm (Cache), nhật ký lỗi, trạng thái phiên làm việc... hãy cài đặt thời gian hết hạn trực tiếp trong cơ sở dữ liệu (ví dụ: tính năng TTL của Redis hay MongoDB). Dữ liệu quá cũ sẽ tự động biến mất, ngăn chặn việc kho lưu trữ phình to theo năm tháng.

## 1. Cơ chế Cuộn ảo (Virtual Scrolling / Windowing)

* Kết quả: Những tin nhắn cũ bị cuộn lên phía trên sẽ lập tức được ẩn đi khỏi cây mã nguồn HTML (gỡ bỏ khỏi DOM) và chỉ được tải lại khi người dùng chủ động cuộn ngược lên. Số lượng phần tử đồ họa luôn giữ ở mức tối thiểu, giúp máy yếu chạy mượt mà 100%.

## 2. Thuật toán Cắt bớt phần tử thừa (DOM Truncation)
Nếu tác giả lười lập trình cơ chế cuộn ảo phức tạp, anh ấy có thể dùng một đoạn code JavaScript ngắn để ép khung chat tự động giới hạn số lượng tin nhắn hiển thị:

* Cách hoạt động: Mỗi khi AI hoặc người dùng gửi một tin nhắn mới, code giao diện sẽ đếm xem tổng số hộp thoại trong khung chat hiện tại là bao nhiêu. Nếu vượt quá ngưỡng quy định (ví dụ: 10 tin nhắn), đoạn code sẽ tự động xóa bỏ phần tử tin nhắn cũ nhất nằm ở trên cùng (chatContainer.firstElementChild.remove()).
* Kết quả: Giao diện web luôn nhẹ nhàng vì không bao giờ phải chứa quá nhiều chữ cùng một lúc.
